# Entry point
Step welcome
    Speak "您好，请问您需要点什么"
    Listen 100
    Branch "食物", foodBranch
    Branch "饮品", drinkBranch
    Branch "套餐", comboBranch
    Branch "查看订单", performCheck
    Branch "退出", exitBranch
    Silence welcome
    Default welcome

# Food Branch
Step foodBranch
    Speak "菜单如下：食物1；食物2；食物3"
    Listen 20
    Branch "食物1", placeFoodOrder
    Branch "食物2", placeFoodOrder
    Branch "食物3", placeFoodOrder
    Branch "返回", welcome
    Silence foodBranch
    Default defaultBranchFood

Step placeFoodOrder
    Action place_order "F"
    Speak "您已下单，您的号码为：$last_order_id"
    Listen 20
    Branch "返回", welcome
    Default defaultBranch

# Check Order Branch
Step checkOrderBranch
    Speak "请输入订单号码"
    Listen 20
    Branch "返回", welcome
    Silence welcome
    Default defaultBranch

Step performCheck
    Speak "请输入订单号码"
    Listen 20
    Action check_order_from_input "user_input"
    Speak "$check_result"
    Listen 20
    Branch "返回", welcome
    Default defaultBranch

Step defaultBranch

    Branch "饮品", drinkBranch
    Branch "套餐", comboBranch
